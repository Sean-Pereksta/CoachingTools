const RANK_PREF_KEY='coachtools.performanceScorecard.ranking.v1';
const DEFAULT_RANK_RULES=Object.freeze([
  {id:'consumer-rate',higher:true},
  {id:'insurance-rate',higher:true},
  {id:'commercial-rate',higher:true},
  {id:'wiper-rate',higher:true},
  {id:'call-quality',higher:true}
]);
function normalizeRankingRules(rules){const seen=new Set(),src=Array.isArray(rules)?rules:DEFAULT_RANK_RULES;return src.filter(r=>r&&r.id&&!seen.has(String(r.id))&&seen.add(String(r.id))).map(r=>({id:String(r.id),higher:r.higher!==false}))}
function loadRankingPrefs(){try{const p=JSON.parse(localStorage.getItem(RANK_PREF_KEY)||'{}');return normalizeRankingRules(p.rules)}catch{return normalizeRankingRules(DEFAULT_RANK_RULES)}}
const rankingState={active:false,rules:loadRankingPrefs(),normalSort:null};
function saveRankingPrefs(){try{localStorage.setItem(RANK_PREF_KEY,JSON.stringify({rules:rankingState.rules}))}catch{}}
function rankingLabel(id){const custom=state.config?.custom?.find(c=>c.id===id);return BUILTINS[id]?.label||custom?.label||id}
function rankingAvailableIds(){const dept=$('departmentSel')?.value||'All',built=Object.keys(BUILTINS).filter(id=>id!=='representative').filter(id=>!(BUILTINS[id]?.dept==='Retail'&&dept!=='Retail')),custom=(state.config?.custom||[]).map(c=>c.id);return[...new Set([...built,...custom])]}
function rankingValue(row,id){const value=sortValue(row,id);return Number.isFinite(value)?value:NaN}
function rankingRuleApplies(rule,rows){const dept=$('departmentSel')?.value||'All',builtin=BUILTINS[rule.id];if(builtin?.dept==='Retail'&&dept!=='Retail')return false;return rows.some(row=>Number.isFinite(rankingValue(row,rule.id)))}
function rankPosition(value,values,higher){if(!Number.isFinite(value))return null;return 1+values.filter(v=>higher?v>value:v<value).length}
function buildRanking(allRows){const rules=rankingState.rules.filter(rule=>rankingRuleApplies(rule,allRows)),pools=new Map(rules.map(rule=>[rule.id,allRows.map(row=>rankingValue(row,rule.id)).filter(Number.isFinite)])),items=allRows.map(row=>{const details=rules.map(rule=>{const value=rankingValue(row,rule.id),rank=rankPosition(value,pools.get(rule.id)||[],rule.higher);return{...rule,value,rank}}),complete=rules.length>0&&details.every(d=>Number.isFinite(d.rank));return{row,details,score:complete?details.reduce((sum,d)=>sum+d.rank,0):NaN,overall:NaN}}),scores=items.map(x=>x.score).filter(Number.isFinite);for(const item of items)if(Number.isFinite(item.score))item.overall=1+scores.filter(score=>score<item.score).length;return{rules,items}}
function rankingValueText(row,id,value){if(!Number.isFinite(value))return'No data';if(row.packs[id])return pct(value);if(id==='days-since-coaching')return`${Math.round(value)}d`;if(id==='kpi-history')return`${Math.round(value)}w`;return int(value)}
function rankingSortValue(item,key){if(key==='representative')return item.row.rep.displayName;if(key==='rank-overall')return item.overall;if(key==='rank-score')return item.score;if(String(key).startsWith('rank:')){const id=String(key).slice(5);return item.details.find(d=>d.id===id)?.rank??NaN}return NaN}
function sortRankingItems(items){const k=state.sort.key,d=state.sort.dir;return items.sort((a,b)=>{let av=rankingSortValue(a,k),bv=rankingSortValue(b,k);if(typeof av==='string')return av.localeCompare(String(bv))*d;if(!Number.isFinite(av))av=d>0?Infinity:-Infinity;if(!Number.isFinite(bv))bv=d>0?Infinity:-Infinity;return(av-bv)*d})}
function populateRankingSort(rules){const options=[['rank-overall','Overall Rank'],['rank-score','Rank Score'],...rules.map(r=>[`rank:${r.id}`,`${rankingLabel(r.id)} Rank`]),['representative','Representative']];if(!options.some(([id])=>id===state.sort.key)){state.sort.key='rank-overall';state.sort.dir=1}$('sortSel').innerHTML=options.map(([id,label])=>`<option value="${esc(id)}">${esc(label)}</option>`).join('');$('sortSel').value=state.sort.key}
function renderRankingTable(visibleRows,totalRows,allRows){const ranking=buildRanking(allRows),visibleIds=new Set(visibleRows.map(r=>r.rep.personId)),items=sortRankingItems(ranking.items.filter(x=>visibleIds.has(x.row.rep.personId)));populateRankingSort(ranking.rules);const arrow=id=>state.sort.key===id?(state.sort.dir>0?' ↑':' ↓'):'';$('tableHead').innerHTML=`<tr><th class="repSticky" data-rank-sort="representative">Representative${arrow('representative')}</th><th data-rank-sort="rank-overall">Overall Rank${arrow('rank-overall')}</th><th data-rank-sort="rank-score">Rank Score${arrow('rank-score')}</th>${ranking.rules.map(rule=>`<th data-rank-sort="rank:${esc(rule.id)}">${esc(rankingLabel(rule.id))} Rank${arrow(`rank:${rule.id}`)}</th>`).join('')}</tr>`;$('tableBody').innerHTML=items.length?items.map(item=>`<tr><td class="repSticky">${renderCell(item.row,'representative')}</td><td class="metricCell"><div class="metricMain">${Number.isFinite(item.overall)?`#${item.overall}`:'—'}</div><div class="metricMeta"><span>Lowest score ranks #1</span></div></td><td class="metricCell"><div class="metricMain">${Number.isFinite(item.score)?int(item.score):'—'}</div><div class="metricMeta"><span>${Number.isFinite(item.score)?'Sum of metric ranks':'Incomplete ranking data'}</span></div></td>${item.details.map(detail=>`<td class="metricCell"><div class="metricMain">${Number.isFinite(detail.rank)?`#${detail.rank}`:'—'}</div><div class="metricVolume">${esc(rankingValueText(item.row,detail.id,detail.value))}</div><div class="metricMeta"><span>${detail.higher?'Highest is best':'Lowest is best'}</span></div></td>`).join('')}</tr>`).join(''):`<tr><td colspan="${3+ranking.rules.length}" class="profileEmpty">No representatives match this ranking view.</td></tr>`;const shown=visibleRows.length===totalRows?`${visibleRows.length} representatives`:`${visibleRows.length} of ${totalRows} representatives`;$('workspaceMeta').textContent=`${shown} · Ranking Scorecard · ${selectedWindowLabel()} · lower Rank Score is better`}
function rankingRuleRows(){return rankingState.rules.map(rule=>`<div class="customRow"><div><b>${esc(rankingLabel(rule.id))}</b><small>${rule.higher?'Highest is best':'Lowest is best'}</small></div><div style="display:flex;gap:6px;align-items:center"><select class="rankDirection" data-rank-direction="${esc(rule.id)}"><option value="high" ${rule.higher?'selected':''}>Highest is best</option><option value="low" ${!rule.higher?'selected':''}>Lowest is best</option></select><button class="btn small" data-rank-remove="${esc(rule.id)}">Remove</button></div></div>`).join('')}
function renderRankingConfig(){const box=$('rankingConfigBox');if(!box||!state.config)return;const used=new Set(rankingState.rules.map(r=>r.id)),available=rankingAvailableIds().filter(id=>!used.has(id));box.innerHTML=`<h3>Ranking Scorecard</h3><div class="workspaceMeta" style="margin-bottom:8px">Each metric contributes its rank number. #1 + #1 + #1 = Rank Score 3. Lowest total score is Overall Rank #1.</div><div class="customRows">${rankingState.rules.length?rankingRuleRows():'<div class="workspaceMeta">No ranking metrics selected.</div>'}</div><div class="fieldGrid" style="margin-top:10px"><div class="field"><label>Add score</label><select id="rankAddMetric">${available.map(id=>`<option value="${esc(id)}">${esc(rankingLabel(id))}</option>`).join('')}</select></div><div class="field"><label>Direction</label><select id="rankAddDirection"><option value="high">Highest is best</option><option value="low">Lowest is best</option></select></div></div><div style="display:flex;gap:6px;flex-wrap:wrap;margin-top:9px"><button class="btn primary small" id="rankAddBtn" type="button" ${available.length?'':'disabled'}>Add score</button><button class="btn small" id="rankResetBtn" type="button">Reset defaults</button><button class="btn small" id="rankClearBtn" type="button">Clear all</button></div><div class="workspaceMeta" style="margin-top:8px">Tip: create a custom Checklist column such as Correctives above, then add it here and choose Lowest is best.</div>`}
function syncRankingViewButton(){const btn=$('rankingViewBtn');if(!btn)return;btn.textContent=rankingState.active?'▦ Team Scorecard':'🏆 Ranking Scorecard';btn.classList.toggle('primary',rankingState.active);btn.setAttribute('aria-pressed',rankingState.active?'true':'false')}
function setRankingActive(active){if(Boolean(active)===rankingState.active)return;if(active){rankingState.normalSort={...state.sort};state.sort={key:'rank-overall',dir:1}}else state.sort=rankingState.normalSort||{key:'representative',dir:1};rankingState.active=Boolean(active);render()}
function installRankingScorecard(){const tools=document.querySelector('#scorecardWorkspace .workspaceTools');if(tools&&!$('rankingViewBtn')){const btn=document.createElement('button');btn.className='btn small icon';btn.id='rankingViewBtn';btn.type='button';btn.setAttribute('aria-pressed','false');btn.onclick=()=>setRankingActive(!rankingState.active);tools.insertBefore(btn,$('fullscreenBtn'))}const body=document.querySelector('#drawer .drawerBody');if(body&&!$('rankingConfigBox')){const box=document.createElement('section');box.className='box';box.id='rankingConfigBox';box.innerHTML='<h3>Ranking Scorecard</h3><div class="workspaceMeta">Ranking controls load with scorecard data.</div>';body.appendChild(box)}const style=document.createElement('style');style.textContent='#rankingConfigBox .rankDirection{border:1px solid var(--line);border-radius:8px;background:var(--panel);padding:6px;font-size:9px}th[data-rank-sort]{cursor:pointer}th[data-rank-sort]:hover{color:var(--ink)}';document.head.appendChild(style);$('sortSel').addEventListener('change',e=>{if(!rankingState.active)return;e.stopImmediatePropagation();state.sort.key=e.target.value;state.sort.dir=1;render()},true);document.addEventListener('click',e=>{const sort=e.target.closest('[data-rank-sort]');if(sort){e.preventDefault();const key=sort.dataset.rankSort;if(state.sort.key===key)state.sort.dir*=-1;else{state.sort.key=key;state.sort.dir=1}render();return}const remove=e.target.closest('[data-rank-remove]');if(remove){rankingState.rules=rankingState.rules.filter(r=>r.id!==remove.dataset.rankRemove);saveRankingPrefs();render();return}if(e.target.id==='rankAddBtn'){const metric=$('rankAddMetric')?.value;if(metric){rankingState.rules.push({id:metric,higher:$('rankAddDirection')?.value!=='low'});rankingState.rules=normalizeRankingRules(rankingState.rules);saveRankingPrefs();render()}return}if(e.target.id==='rankClearBtn'){rankingState.rules=[];saveRankingPrefs();render();return}if(e.target.id==='rankResetBtn'){rankingState.rules=normalizeRankingRules(DEFAULT_RANK_RULES);saveRankingPrefs();render();return}});document.addEventListener('change',e=>{const select=e.target.closest('[data-rank-direction]');if(!select)return;const rule=rankingState.rules.find(r=>r.id===select.dataset.rankDirection);if(!rule)return;rule.higher=select.value!=='low';saveRankingPrefs();render()});syncRankingViewButton()}
function render(){const allRows=getRows(),rows=allRows.filter(r=>matchesQuickFilter(r));renderSummary(allRows);renderQuickFilters(allRows);if(rankingState.active)renderRankingTable(rows,allRows.length,allRows);else renderTable(rows,allRows.length);syncRankingViewButton();renderRankingConfig()}
installRankingScorecard();
