icon';btn.id='rankingViewBtn';btn.type='button';btn.setAttribute('aria-pressed','false');btn.onclick=()=>setRankingActive(!rankingState.active);tools.insertBefore(btn,$('scorecardExportMenu')||$('fullscreenBtn'))}const body=document.querySelector('#drawer .drawerBody');if(body&&!$('rankingConfigBox')){const box=document.createElement('section');box.className='box';box.id='rankingConfigBox';box.innerHTML='<h3>Ranking Scorecard</h3><div class="workspaceMeta">Ranking controls load with scorecard data.</div>';body.appendChild(box)}installScorecardExtraStyles();$('sortSel').addEventListener('change',e=>{if(rankingState.active){e.stopImmediatePropagation();state.sort.key=e.target.value;state.sort.dir=1;render();return}if(e.target.value==='rank-overall'){e.stopImmediatePropagation();state.sort.key='rank-overall';state.sort.dir=1;render()}},true);document.addEventListener('click',e=>{const normalRank=e.target.closest('[data-normal-rank-sort]');if(normalRank){e.preventDefault();if(state.sort.key==='rank-overall')state.sort.dir*=-1;else{state.sort.key='rank-overall';state.sort.dir=1}render();return}const sort=e.target.closest('[data-rank-sort]');if(sort){e.preventDefault();const key=sort.dataset.rankSort;if(state.sort.key===key)state.sort.dir*=-1;else{state.sort.key=key;state.sort.dir=1}render();return}const remove=e.target.closest('[data-rank-remove]');if(remove){rankingState.rules=rankingState.rules.filter(r=>r.id!==remove.dataset.rankRemove);saveRankingPrefs();render();return}if(e.target.id==='rankAddBtn'){const metric=$('rankAddMetric')?.value;if(metric){rankingState.rules.push({id:metric,higher:$('rankAddDirection')?.value!=='low'});rankingState.rules=normalizeRankingRules(rankingState.rules);saveRankingPrefs();render()}return}if(e.target.id==='rankClearBtn'){rankingState.rules=[];saveRankingPrefs();render();return}if(e.target.id==='rankResetBtn'){rankingState.rules=normalizeRankingRules(DEFAULT_RANK_RULES);saveRankingPrefs();render();return}});document.addEventListener('change',e=>{const select=e.target.closest('[data-rank-direction]');if(!select)return;const rule=rankingState.rules.find(r=>r.id===select.dataset.rankDirection);if(!rule)return;rule.higher=select.value!=='low';saveRankingPrefs();render()});syncRankingViewButton();syncDisplayModeControl()}
